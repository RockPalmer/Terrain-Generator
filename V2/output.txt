PERLIN_NOISE(size,seed,scale,octaves):
	noise_map = Image(size)
	for y in range(size):
		for x in range(size):
			value = 0
			amplitude = 1
			frequency = 1
			amplitude_sum = 0
			for _ in range(octaves):
				angle_x = tau * x / size
				angle_y = tau * y / size
				nx = cos(angle_x) * scale * frequency
				ny = sin(angle_x) * scale * frequency
				nz = cos(angle_y) * scale * frequency
				nw = sin(angle_y) * scale * frequency + seed
				sample = perlin5(nx,ny,nz,nw,0)
				value += sample * amplitude
				amplitude_sum += amplitude
				amplitude *= 0.5
				frequency *= 2
			noise_map[x,y] = value / amplitude_sum
	return noise_map
