true_surface ~ C
surface_related_pressure_a ~ C
surface_related_pressure ~ C
sunlight_related_pressure = OVERWORLD_SUNLIGHT_MAX - sunlight(t)
pressure_diff = NOISE * sunlight_related_pressure
pressure = pressure_diff + surface_related_pressure
inv_pressure = OVERWORLD_PRESSURE_MAX - pressure
humidity = (
	inv_pressure + temperature + water_ratio
)/3
surface ~ C
distance ~ C
temperature = ifMap(
	(sunlight(t) + distance.max - distance)/2,
	land,
	(sunlight(t) + distance.max - distance)/2 - OVERWORLD_TEMPERATURE_LAND_DIFFERENCE,
)
rain = (temperature + humidity)/2
